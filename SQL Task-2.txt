    WHEN price>700 AND stock_quantity<20

        THEN 'Low Stock'
    ELSE 'Normal'

FROM Books;

CREATE PROCEDURE GetMemberBorrowDetails(IN mid INT)
BEGIN
    SELECT m.member_name,
           b.book_name,
           br.borrow_data,
           br.return_date
    FROM Members m
    JOIN Borrow br ON m.member_id=br.member_id
    JOIN Books b ON br.book_id=b.book_id
    WHERE m.member_id=mid;
END //
